		<STSource>(*=====================================================*)
(*   POINTS DE FREINAGE ROTATION                       *)
(*   Ralentissement 10° avant chaque position          *)
(*=====================================================*)

(*POINT DE FREINAGE ROTATION*)
IF (Rota.rRelativePos &lt;=10.0 AND Rota.rRelativePos &gt;=0.0 AND ATV_3A1.RUNRV) OR (Rota.rRelativePos &gt;=80.0 AND ATV_3A1.RUNFW) OR 
(Rota.rRelativePos &lt;=-80.0 AND Rota.rRelativePos &gt;=-90.0 AND ATV_3A1.RUNRV) OR (Rota.rRelativePos&lt;=-170.0 AND ATV_3A1.RUNRV) OR
(Rota.rRelativePos &gt;=-10.0 AND Rota.rRelativePos &lt;=0.0 AND ATV_3A1.RUNFW) THEN
bSLOWRot:= TRUE;
ELSE bSLOWRot:= FALSE;
END_IF;


(*=====================================================*)
(*   CHANGEMENT RATIO QUAND FREINAGE                   *)
(*   Ralentissement vitesse / 2                        *)
(*=====================================================*)
IF bSLOWRot THEN
	ATV_3A1.RATIO:=0.3;
ELSE
	ATV_3A1.RATIO:=1.0;
END_IF;
