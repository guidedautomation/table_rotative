(* APPRENTISSAGE DES POSITIONS - AXE X TRANSLATION CAGE                         *)
(* ============================================================================ *)

(* ---------------------------------------------------------------------------- *)
(* TEACH POINT 0mm - PRISE D'ORIGINE                                            *)
(* ---------------------------------------------------------------------------- *)
IF hTP_POSX THEN
    PosX.HomePosition := PosX.Value;           (* Mémorisation position capteur = origine *)
    PosX.bHOMED := TRUE;                       (* Axe référencé ✓ *)
END_IF;

(* ---------------------------------------------------------------------------- *)
(* CALCUL POSITION RELATIVE (corrigé)                                           *)
(* ---------------------------------------------------------------------------- *)
IF PosX.bHOMED THEN
    PosX.RelativePos := PosX.Value - PosX.HomePosition;  (* Position relative à l'origine *)
ELSE
    PosX.RelativePos := 0;                               (* Pas de calcul si pas référencé *)
END_IF;


(* ============================================================================ *)
(* APPRENTISSAGE DES POSITIONS - AXE Y M/D CAGE                                 *)
(* ============================================================================ *)

(* ---------------------------------------------------------------------------- *)
(* TEACH POINT 0mm - PRISE D'ORIGINE                                            *)
(* ---------------------------------------------------------------------------- *)
IF hTP_POSY THEN
    PosY1.HomePosition := PosY1.Value;           (* Mémorisation position capteur = origine *)
    PosY2.HomePosition := PosY2.Value;
    PosY1.bHOMED := TRUE;                        (* Axe référencé ✓ *)
    PosY2.bHOMED := TRUE;
END_IF;

(* ----------------------------------------------------------------------------*)
(* CALCUL POSITION RELATIVE (corrigé)                                          *)
(* ----------------------------------------------------------------------------*)
IF PosY1.bHOMED THEN
    PosY1.RelativePos := PosY1.Value - PosY1.HomePosition;  (* Position relative à l'origine *)
ELSE
    PosY1.RelativePos := 0;                                 (* Pas de calcul si pas référencé *)
END_IF;

IF PosY2.bHOMED THEN
    PosY2.RelativePos := PosY2.Value - PosY2.HomePosition;  (* Position relative à l'origine *)
ELSE
    PosY2.RelativePos := 0;                                 (* Pas de calcul si pas référencé *)
END_IF;


(* ============================================================================ *)
(* APPRENTISSAGE DES POSITIONS - ROTATION TRANSLATION CAGE                      *)
(* ============================================================================ *)

(* ---------------------------------------------------------------------------- *)
(* TEACH POINT 0° - PRISE D'ORIGINE                                             *)
(* ---------------------------------------------------------------------------- *)
IF hTP_ROTA THEN
    Rota.rHomePosition := rAngle;           (* Mémorisation position capteur = origine *)
    Rota.bHOMED := TRUE;                       (* Axe référencé ✓ *)
END_IF;

(* ---------------------------------------------------------------------------- *)
(* CALCUL POSITION RELATIVE (corrigé)                                           *)
(* ---------------------------------------------------------------------------- *)
IF Rota.bHOMED THEN
    Rota.rRelativePos := Rota.rHomePosition - rAngle ;  (* Position relative à l'origine *)
ELSE
    Rota.rRelativePos := 0.0;                               (* Pas de calcul si pas référencé *)
END_IF;

(*=====================================================*)
(*   TEACH POSITIONS (prise d'origine)                 *)
(*=====================================================*)

IF hTP_ROTA THEN
    
    (* Pos 0 : 0° (position actuelle = origine) *)
    Rota.SET_POS[0] := REAL_TO_DINT(Rota.rRelativePos);
    
    (* Pos 1 : +90° *)
    Rota.SET_POS[1] := Rota.SET_POS[0] + 90;
    
    (* Pos 2 : -90° *)
    Rota.SET_POS[3] := Rota.SET_POS[0] - 90;
    
    (* Pos 3 : -180° *)
    Rota.SET_POS[2] := Rota.SET_POS[0] - (90 * 2);
    
   
END_IF;
