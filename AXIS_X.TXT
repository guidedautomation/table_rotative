(*SORTIE AXE X*)
PosX.Value		:= (PosX.rawValue/1000);

(*POINT DE FREINAGE AXE X*)
IF (PosX.RelativePos&lt;=100 AND ATV_4A1.RUNRV) OR (PosX.RelativePos &gt;=600 AND ATV_4A1.RUNFW) OR
((PosX.RelativePos&gt;250 AND PosX.RelativePos&lt;=370)  AND ATV_4A1.RUNFW) OR
((PosX.RelativePos&lt;400 AND PosX.RelativePos&gt;=350) AND ATV_4A1.RUNRV) THEN
bSLOWX:= TRUE;
ELSE bSLOWX:= FALSE;
END_IF;

IF bSLOWX
THEN ATV_4A1.RATIO:=0.3;
ELSE
ATV_4A1.RATIO:=1.0;
END_IF;

			<STSource>(* ============================================================================ *)
(* DFB_POSITION_DETECTOR - Détection de positions uniquement                  *)
(* Pas de gestion mouvement - Juste détection fenêtres                        *)
(* ============================================================================ *)




(* ============================================================================ *)
(* LOGIQUE DE DÉTECTION                                                        *)
(* ============================================================================ *)

(* Détection mouvement *)
Moving := RunFW OR RunRV;

(* Détection de toutes les positions *)
AnyPositionReached := FALSE;
CurrentPosition := -1;

FOR i := 0 TO (NbTargets - 1) DO
    (* Vérification fenêtre de tolérance *)
    InPosition[i] := 
        (ActualPosition &gt;= (TargetPositions[i] - Tolerance)) AND 
        (ActualPosition &lt;= (TargetPositions[i] + Tolerance));
    
    (* Mémorisation position actuelle *)
    IF InPosition[i] THEN
        AnyPositionReached := TRUE;
        CurrentPosition := i;
    END_IF;
END_FOR;
